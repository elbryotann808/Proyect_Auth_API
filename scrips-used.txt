# Project initiation
npm init -y

# Configuration with TypeScript
npm i typescript @types/node tsx -D
npx tsc --init

# Cofiguration with esLint
npm i eslint -D
npx eslint --init 

# Server dependencies
npm i express cors
npm i @types/express @types/cors -D

# Documentation dependencies
npm i swagger-ui-express
npm i -D @types/swagger-ui-express

# Testing dependencies
npm i -D vitest supertest @types/supertest

# Database dependencies
npm i prisma -D
npm i @prisma/client
npx prisma init
npx prisma migrate dev --name "init"
npx prisma studio

# Security dependencies
npm i bcrypt
npm i jsonwebtoken
npm i cookie-parser

npm i -D @types/bcrypt @types/jsonwebtoken @types/cookie-parser
